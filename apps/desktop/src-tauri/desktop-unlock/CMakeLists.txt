cmake_minimum_required(VERSION 3.24)
project(desktop_unlock LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(desktop-unlock SHARED provider.cpp credential.cpp mailbox.cpp serialization.cpp exports.def)
target_compile_definitions(desktop-unlock PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(desktop-unlock PRIVATE /W4 /WX /guard:cf /utf-8)
target_link_options(desktop-unlock PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(desktop-unlock PRIVATE advapi32 crypt32 credui ole32 shell32 shlwapi secur32)
set_target_properties(desktop-unlock PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${UNLOCK_OUTPUT}")
enable_testing()
add_executable(unlock-tests tests.cpp provider_tests.cpp provider.cpp credential.cpp mailbox.cpp serialization.cpp)
target_compile_definitions(unlock-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNLOCK_TESTING)
target_compile_options(unlock-tests PRIVATE /W4 /WX /utf-8)
target_link_libraries(unlock-tests PRIVATE advapi32 crypt32 credui ole32 shell32 shlwapi secur32)
add_test(NAME unlock-tests COMMAND unlock-tests)
